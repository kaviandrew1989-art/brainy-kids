let pIndex=0,rIndex=0,readingScore=0;
function startReading(){
if(!name){showLogin();return}
hideAll();
document.getElementById("reading").style.display="block";
pIndex=0;rIndex=0;readingScore=0;
showReading();
}
function showReading(){
let p=passages[pIndex];
document.getElementById("passageCount").innerText=
"Passage "+(pIndex+1)+" of 5 | Question "+(rIndex+1)+" of 6";
document.getElementById("passageTitle").innerText=p.title;
document.getElementById("passageText").innerText=p.text;
let q=p.qs[rIndex];
document.getElementById("readingQuestion").innerHTML="<h3>"+q[0]+"</h3>";
document.getElementById("readingOptions").innerHTML="";
document.getElementById("readingFeedback").innerText="";
document.getElementById("readingNext").style.display="none";
[...q[2],q[1]].sort(()=>Math.random()-.5).forEach(opt=>{
let b=document.createElement("button");
b.className="option";b.innerText=opt;
b.onclick=()=>readingAnswer(opt,b);
document.getElementById("readingOptions").appendChild(b);
});
}
function readingAnswer(value,button){
let q=passages[pIndex].qs[rIndex];
document.querySelectorAll("#readingOptions button").forEach(b=>b.disabled=true);
if(value===q[1]){
readingScore++;stars++;completed++;save();updateStats();
document.getElementById("readingFeedback").innerText="🎉 Correct! +1 star!";
}else{
button.classList.add("wrong");
document.getElementById("readingFeedback").innerText="Correct answer: "+q[1];
}
document.getElementById("readingNext").style.display="inline-block";
}
function nextReading(){
rIndex++;
if(rIndex===6){
pIndex++;rIndex=0;
if(pIndex===5){
document.getElementById("passageTitle").innerText="🏆 Reading Challenge Completed!";
document.getElementById("passageText").innerText=
"Wonderful work! You completed all five passages and 30 questions. Your score: "+readingScore+"/30";
document.getElementById("readingQuestion").innerText="";
document.getElementById("readingOptions").innerHTML="";
document.getElementById("readingFeedback").innerText="Keep reading and keep growing! 🦉";
document.getElementById("readingNext").style.display="none";
return;
}
}
showReading();
}
